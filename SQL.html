CREATE TABLE Employees (
    employee_id INT IDENTITY(1,1) PRIMARY KEY,
    first_name VARCHAR(50) NOT NULL,
    last_name VARCHAR(50) NOT NULL,
    role VARCHAR(50) DEFAULT 'Therapist',
    specialty VARCHAR(100),
    commission_rate DECIMAL(5,2) DEFAULT 0.50,
    hire_date DATE DEFAULT GETDATE()
);

CREATE TABLE Customers (
    customer_id INT IDENTITY(1,1) PRIMARY KEY,
    first_name VARCHAR(50) NOT NULL,
    last_name VARCHAR(50) NOT NULL,
    phone VARCHAR(20),
    email VARCHAR(100),
    join_date DATE DEFAULT GETDATE()
);

CREATE TABLE Services (
    service_id INT IDENTITY(1,1) PRIMARY KEY,
    service_name VARCHAR(100) NOT NULL,
    duration_minutes INT NOT NULL,
    price DECIMAL(10,2) NOT NULL
);

CREATE TABLE Appointments (
    appointment_id INT IDENTITY(1,1) PRIMARY KEY,
    customer_id INT NOT NULL,
    employee_id INT NOT NULL,
    service_id INT NOT NULL,
    appointment_datetime DATETIME NOT NULL,
    status VARCHAR(20) DEFAULT 'Scheduled',
    FOREIGN KEY (customer_id) REFERENCES Customers(customer_id),
    FOREIGN KEY (employee_id) REFERENCES Employees(employee_id),
    FOREIGN KEY (service_id) REFERENCES Services(service_id)
);

CREATE TABLE Payments (
    payment_id INT IDENTITY(1,1) PRIMARY KEY,
    appointment_id INT NOT NULL,
    employee_id INT NOT NULL,
    service_amount DECIMAL(10,2) DEFAULT 0,
    product_amount DECIMAL(10,2) DEFAULT 0,
    tip_amount DECIMAL(10,2) DEFAULT 0,
    tax_amount DECIMAL(10,2) DEFAULT 0,
    total_paid DECIMAL(10,2) NOT NULL,
    payment_method VARCHAR(30),
    payment_date DATETIME DEFAULT GETDATE(),
    FOREIGN KEY (appointment_id) REFERENCES Appointments(appointment_id),
    FOREIGN KEY (employee_id) REFERENCES Employees(employee_id)
);

CREATE VIEW vw_DailyFinancialSummary AS
SELECT
    CAST(payment_date AS DATE) AS business_date,
    SUM(service_amount) AS service_sales_before_tax,
    SUM(product_amount) AS product_sales_before_tax,
    SUM(service_amount + product_amount) AS subtotal_before_tax,
    SUM(tax_amount) AS total_tax_collected,
    SUM(tip_amount) AS total_tips,
    SUM(total_paid) AS total_money_collected
FROM Payments
GROUP BY CAST(payment_date AS DATE);

GO

CREATE VIEW vw_YearlyTaxSummary AS
SELECT
    YEAR(payment_date) AS tax_year,
    SUM(service_amount + product_amount) AS total_sales_before_tax,
    SUM(ROUND(tax_amount, 2)) AS total_tax_collected,
    SUM(tip_amount) AS total_tips_paid,
    SUM(total_paid) AS total_revenue
FROM Payments
GROUP BY YEAR(payment_date);

GO

CREATE VIEW vw_EndOfDayEmployeeSummary AS
SELECT
    CAST(p.payment_date AS DATE) AS business_date,
    e.employee_id,
    e.first_name,
    e.last_name,
    SUM(p.service_amount) AS service_sales_before_tax,
    SUM(p.product_amount) AS product_sales_before_tax,
    SUM(p.tax_amount) AS taxes_collected,
    SUM(p.tip_amount) AS tips_earned,
    SUM(p.total_paid) AS total_collected_with_tax_and_tips
FROM Payments p
JOIN Employees e ON p.employee_id = e.employee_id
GROUP BY CAST(p.payment_date AS DATE), e.employee_id, e.first_name, e.last_name;

GO

CREATE VIEW vw_PayrollSummary AS
SELECT
    e.employee_id,
    e.first_name,
    e.last_name,
    SUM(p.service_amount * e.commission_rate) AS service_commission,
    SUM(p.tip_amount) AS tips,
    SUM(p.service_amount * e.commission_rate) + SUM(p.tip_amount) AS total_pay
FROM Payments p
JOIN Employees e ON p.employee_id = e.employee_id
GROUP BY e.employee_id, e.first_name, e.last_name;

GO

CREATE PROCEDURE sp_BookAppointment
    @customer_id INT,
    @employee_id INT,
    @service_id INT,
    @appointment_datetime DATETIME
AS
BEGIN
    INSERT INTO Appointments (customer_id, employee_id, service_id, appointment_datetime, status)
    VALUES (@customer_id, @employee_id, @service_id, @appointment_datetime, 'Scheduled');
END;

GO

CREATE PROCEDURE sp_ProcessPayment
    @appointment_id INT,
    @employee_id INT,
    @service_amount DECIMAL(10,2),
    @product_amount DECIMAL(10,2),
    @tip_amount DECIMAL(10,2),
    @payment_method VARCHAR(30)
AS
BEGIN
    DECLARE @tax_amount DECIMAL(10,2);
    DECLARE @total_paid DECIMAL(10,2);

    SET @tax_amount = ROUND((@service_amount + @product_amount) * 0.0825, 2);
    SET @total_paid = @service_amount + @product_amount + @tax_amount + @tip_amount;

    INSERT INTO Payments
    (appointment_id, employee_id, service_amount, product_amount, tip_amount, tax_amount, total_paid, payment_method, payment_date)
    VALUES
    (@appointment_id, @employee_id, @service_amount, @product_amount, @tip_amount, @tax_amount, @total_paid, @payment_method, GETDATE());
END;

GO

CREATE PROCEDURE sp_CancelAppointment
    @appointment_id INT
AS
BEGIN
    UPDATE Appointments
    SET status = 'Cancelled'
    WHERE appointment_id = @appointment_id;
END;

GO

CREATE PROCEDURE sp_GeneratePayroll
AS
BEGIN
    SELECT
        e.employee_id,
        e.first_name,
        e.last_name,
        SUM(p.service_amount * e.commission_rate) AS service_commission,
        SUM(p.tip_amount) AS tips,
        SUM(p.service_amount * e.commission_rate) + SUM(p.tip_amount) AS total_pay
    FROM Payments p
    JOIN Employees e ON p.employee_id = e.employee_id
    GROUP BY e.employee_id, e.first_name, e.last_name;
END;

GO
